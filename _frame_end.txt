  var colors = ['rgba(238,21,21,', 'rgba(255,255,255,', 'rgba(183,255,147,'];
  for(var i=0;i<45;i++){
    particles.push({
      x: Math.random()*W, y: Math.random()*H,
      vx: (Math.random()-0.5)*0.6, vy: (Math.random()-0.5)*0.6,
      r: Math.random()*2.5+1,

  });

  function drawParticles(){
    ctx.clearRect(0,0,W,H);
    for(var i=0;i<particles.length;i++){
      var p = particles[i];
      p.x += p.vx; p.y += p.vy;
      if(p.x<0||p.x>W) p.vx*=-1;
      if(p.y<0||p.y>H) p.vy*=-1;
      ctx.beginPath();
      ctx.arc(p.x,p.y,p.r,0,Math.PI*2);
      ctx.fillStyle = p.c;
      ctx.fill();
    }
    // Draw connections near mouse
    for(var i=0;i<particles.length;i++){
      var pi = particles[i];
      var dx = pi.x - mouse.x, dy = pi.y - mouse.y;
      var dist = Math.sqrt(dx*dx+dy*dy);
      if(dist < 120){
        ctx.beginPath();
        ctx.moveTo(pi.x,pi.y);
        ctx.lineTo(mouse.x,mouse.y);
        ctx.strokeStyle = 'rgba(238,21,21,' + (0.08*(1-dist/120)) + ')';
        ctx.lineWidth = 1;
        ctx.stroke();
